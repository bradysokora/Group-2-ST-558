summary_cat <- function(df, group_var, num_var) {
  df %>%
    group_by({{group_var}}) %>%
    summarize(
      n = n(),
      mean = mean({{num_var}}, na.rm = TRUE),
      sd = sd({{num_var}}, na.rm = TRUE),
      median = median({{num_var}}, na.rm = TRUE),
      min = min({{num_var}}, na.rm = TRUE),
      max = max({{num_var}}, na.rm = TRUE),
      .groups = "drop"
    )
}

# Could drop older unstandardize vars
stand_train <- function(df, mean, sd, vars) {
  df %>%
    mutate(
      across(
        all_of(vars),
        function(x) (x - mean[cur_column()]) / sd[cur_column()],
        .names = "standardized_{.col}"
      )
    )
}

bike_train_stan <- bike_train %>%
  mutate(week = factor(if_else(wday(date) %in% c(1, 7), "Weekend", "Weekday"))) %>%
  stand_train(mean_train, sd_train, num_vars)
