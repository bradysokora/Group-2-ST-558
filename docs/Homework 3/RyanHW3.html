Task 4: Writing a Function

We create the getIQR function that determines the interquartile range for a provided vector of numbers.

# Define function
getIQR <- function(x, ...) {                             # ... allows for other arguments to be passed
  quartiles <- quantile(x, probs = c(0.25, 0.75), ...)   # 25th and 75th percentiles
  iqr <- quartiles[[2]] - quartiles[[1]]                 # Removing name on vector and compute IQR
  iqr
}

Creating the getCoefVar function to find the sample coefficient of variation.

# Define function
getCoefVar <- function(x, ...) {
  cv <- sd(x, ...) / abs(mean(x, ...))  # Sample coeff of variation equation
  cv
}

Creating the getCoefVar function to return the IQR and coefficient of variation.

# Define function
getSummaries <- function(x) {
  IQR_val <- getIQR(x)
  CV_val <- getCoefVar(x)
  return(list(IQR = IQR_val, CoefVar = CV_val)) # Needed to ensure both values are returned
}
