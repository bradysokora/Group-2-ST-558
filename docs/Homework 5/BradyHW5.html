Writing a function so a user can query the API easily

# Function to query newsapi: takes a search term, a from date, and an api key
query_news <- function(search_term, from_date, api_key) {
  result <- GET(paste0(
    "https://newsapi.org/v2/everything?",
    "q=", search_term, "&",
    "from=", from_date, "&",
    "language=en&",
    "pageSize=50&",
    "apiKey=", api_key
  ))
  parsed <- fromJSON(rawToChar(result$content))
  articles <- as_tibble(parsed$articles)
  return(articles)
}
